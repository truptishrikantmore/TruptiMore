Assignment 1: (function overloading)

Note **: create function name as "calculate"

create four functions in a class

1) create method of return type int
   - add two numbers
2) one of return type int
   to calculate the area of the circle
3) one of return type int 
   area of the rectangle
4) one of return type int 
   area of the square
-----------------------------------------------


import java.util.*;

public class MethodOverloading {

    // addition of numbers
    public int calculate(int a, int b) {
      System.out.println("Addition of two numbers is ::: " + (a + b));
      return (a + b);
    }

    // area of circle
    public int calculate(int radius) { 
      System.out.println("Area of circle is :::" + (int) (3.14 * radius * radius));
      return (int) (3.14 * radius * radius);
    }

    // area of the rectangle
    public int calculate(int length, int breadth, int area) {
      area = length * breadth;
      System.out.println("Area of rectangle is :::" + area);
      return area;
    }

    // area of the square
    public int calculate(double side) {
      System.out.println("Area of square is :::" + (int) (side * side));
      return (int) (side * side);
    }

    public static void main(String[] args) {
      MethodOverloading m = new MethodOverloading();
      m.calculate(10, 20);
      m.calculate(5);
      m.calculate(10, 20, 0);
      m.calculate(4.0);
    }
}
