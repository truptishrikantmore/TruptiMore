Create one object of treeset and add the any 6 programming languages in random order
   - remove two languages
   - add three more languages after removal
   - Check if " Java" is part of your treeset or not
   - remove all the elements from the treeset and display it
   
import java.util.*;

public class Main {
    public static void main(String[] args) {
      
      TreeSet treesetLanguage =  new TreeSet();
      treesetLanguage.add("Java");
      treesetLanguage.add("SQL");
      treesetLanguage.add("C++");
      treesetLanguage.add("Python");
      treesetLanguage.add("C");
      treesetLanguage.add(".Net");
      System.out.println("TreeSet is :: " + treesetLanguage);

      treesetLanguage.remove("C");
      treesetLanguage.remove("C++");
      System.out.println("TreeSet is :: " + treesetLanguage);

      treesetLanguage.add("PHP");
      treesetLanguage.add("JavaScript");
      treesetLanguage.add("React");
      System.out.println("TreeSet is :: " + treesetLanguage);

      boolean isJava = treesetLanguage.contains("Java");
      System.out.println("Contains Java::" + isJava);

      treesetLanguage.clear();
      System.out.println("TreeSet is :: " + treesetLanguage);
    }
}

Use HashMap
Add 10 fruits into it and do the following operations:
1) print map
2) get all keys and values

4) check if map contains "banana" and "apple" fruit

import java.util.*;

public class Main {
    public static void main(String[] args) {
      
      HashMap hm = new HashMap();
      hm.put(1,"Apple");
      hm.put(2,"Banana");
      hm.put(3,"Mango");
      hm.put(4,"Grapes");
      hm.put(5,"Kiwi");
      hm.put(6,"Strawberry");
      hm.put(7,"Orange");
      hm.put(8,"Pineapple");
      hm.put(9,"Lemon");
      hm.put(10,"Leeche");

      System.out.println("Hashmap is :: " + hm);

      hm.remove(5,"Kiwi");
      hm.remove(7);
      System.out.println("Hashmap is :: " + hm);

      System.out.println("Is contains banana :: " + hm.containsValue("Banana") + 
      " >> Is contains apple :: " + hm.containsValue("Apple"));

      System.out.println("Is contains 4th Key :: " + hm.containsKey(4));

      System.out.println("Hashcode of hashmap is :: " + hm.hashCode());

      hm.clear();
      System.out.println("Hashmap is :: " + hm);
    }

}
